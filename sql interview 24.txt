1) A company wants to identify its high-performing departments.
Find the department-wise:
- Total number of employees
- Average salary
- Maximum salary
Display only departments having more than 1 employee.

with dept_emp as (
select d.department_name , count(*) as emp_count , 
avg(e.salary) as avg_salary,
max(e.salary) as max_salary
from employees e
join departments d
on e.department_id = d.department_id
group by d.department_name
)
select department_name , emp_count , avg_salary , max_salary
from dept_emp
where emp_count > 1;

2) A company wants to identify employees who earn more than the average salary of their own department.
Display the employee name, salary, and department name for those employees.

select d.department_name , e.emp_name , e.salary
from employees e
join departments d
on e.department_id = d.department_id
where e.salary > (select avg(e2.salary) from employees e2
                   where e2.department_id = e.department_id);

3) Find duplicate salaries in the employees table and display each salary along with the number of employees earning it.

select e.salary , count(*) as num_of_emp
from employees e
group by e.salary
having count(*) > 1;

4) Find the second-highest distinct salary in the company without using LIMIT.

select emp_name , salary 
from (
select emp_name , salary ,
dense_rank()over(order by salary desc ) as rnk
from employees
)A
where rnk = 2;

5) Find each department's total salary and show only departments whose total salary is greater than the company-wide average department salary.

with dept_salary as (
select d.department_name , 
sum(e.salary) as total_salary
from employees e
join departments d
on e.department_id = d.department_id
group by d.department_name
)
select department_name , total_salary
from dept_salary 
where total_salary > ( select avg(total_salary) from dept_salary);
